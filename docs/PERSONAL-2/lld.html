fx.ts — juice-ийн ГАНЦ хаалга

play(events, settings):
  s ← settings()
  motionOff ← s.reducedMotion ∨ matchMedia('(prefers-reduced-motion: reduce)').matches
  soundOff  ← !s.soundEnabled ∨ s.soundVolume <= 0
  animated ← 0
  for e of events:
     row ← FX_REGISTRY[e.type]; if !row: continue
     announce(row.text(e.data))                       ← ҮРГЭЛЖ (FX-6)
     if !motionOff ∧ animated < MAX_CONCURRENT_FX:    ← FX-7 (3)
        addClass(row.cls); animated++
     if !soundOff: tone(row.hz, s.soundVolume)

// dungeons.ts — ХОЁУЛАА энэ туслахаар дамжина
export function markDungeonPassed(state, dungeonId, at): GameState {
  const done = state.completedDungeonIds.includes(dungeonId)
    ? state.completedDungeonIds : [...state.completedDungeonIds, dungeonId];
  return { ...state, completedDungeonIds: done,
           dungeonStats: { ...state.dungeonStats,
                           [dungeonId]: { lastPassedDate: dayOf(at) } } };
}

⚠ lastPassedDate нь дахин тэнцэх бүрд шинэчлэгдэнэ (XP 0 байсан ч) — refresher нь «хамгийн сүүлд хэзээ хүрсэн»-ийг хэмждэг. Шинэ архитектурын дүрэм: dungeonStats: бичилт shared/core/** дотор ЗӨВХӨН dungeons.ts ба saves.ts(newGame)-д олдоно.
